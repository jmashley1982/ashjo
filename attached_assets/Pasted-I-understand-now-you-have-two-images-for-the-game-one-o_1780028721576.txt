Step 2: Create popup.js with the following complete code. This script creates the popup logic, game state, sound synthesis, and DOM manipulation.

```javascript
// popup.js
(function() {
  // Prevent re-trigger on internal navigation
  if (sessionStorage.getItem('ashjo_debt_popup') === 'dismissed') return;

  // ---------- Sound Synthesis (Web Audio API) ----------
  function playCashSound() {
    const AudioContext = window.AudioContext || window.webkitAudioContext;
    if (!AudioContext) return;
    const ctx = new AudioContext();
    const now = ctx.currentTime;
    [800, 1200].forEach((freq, i) => {
      const osc = ctx.createOscillator();
      const gain = ctx.createGain();
      osc.type = 'square';
      osc.frequency.setValueAtTime(freq, now + i * 0.1);
      gain.gain.setValueAtTime(0.3, now + i * 0.1);
      gain.gain.exponentialRampToValueAtTime(0.001, now + i * 0.1 + 0.15);
      osc.connect(gain);
      gain.connect(ctx.destination);
      osc.start(now + i * 0.1);
      osc.stop(now + i * 0.1 + 0.15);
    });
  }

  // ---------- Create popup HTML dynamically ----------
  const popupHTML = `
    <div id="debt-popup">
      <div class="popup-inner">
        <button class="close-btn" id="popup-close-btn">&times;</button>
        <div id="popup-content"></div>
      </div>
    </div>
  `;
  document.body.insertAdjacentHTML('beforeend', popupHTML);
  const popup = document.getElementById('debt-popup');
  const content = document.getElementById('popup-content');
  const closeBtn = document.getElementById('popup-close-btn');

  // ---------- State ----------
  let currentScreen = 'askDebt';

  // ---------- Helper to close popup and remember ----------
  function closePopup() {
    sessionStorage.setItem('ashjo_debt_popup', 'dismissed');
    if (popup) popup.remove();
  }

  closeBtn.addEventListener('click', closePopup);

  // ---------- Render screens ----------
  function render() {
    switch (currentScreen) {
      case 'askDebt':
        content.innerHTML = `
          <h2>Do you owe Ash money?</h2>
          <div class="btn-group">
            <button id="btn-yes-debt">YES</button>
            <button id="btn-no-debt" class="secondary">NO</button>
          </div>
        `;
        document.getElementById('btn-yes-debt').addEventListener('click', () => {
          currentScreen = 'askWork';
          render();
        });
        document.getElementById('btn-no-debt').addEventListener('click', () => {
          currentScreen = 'debtResponse';
          render();
        });
        break;

      case 'debtResponse':
        content.innerHTML = `
          <h2>Better not.</h2>
          <p>(closing soon...)</p>
        `;
        setTimeout(closePopup, 2000);
        break;

      case 'askWork':
        content.innerHTML = `
          <h2>Wanna work off some of the debt?</h2>
          <div class="btn-group">
            <button id="btn-yes-work">YES</button>
            <button id="btn-no-work" class="secondary">NO</button>
          </div>
        `;
        document.getElementById('btn-yes-work').addEventListener('click', () => {
          currentScreen = 'game';
          render();
        });
        document.getElementById('btn-no-work').addEventListener('click', () => {
          currentScreen = 'workNoResponse';
          render();
        });
        break;

      case 'workNoResponse':
        content.innerHTML = `
          <h2>Pfft, well fine. Better pay up soon, loser.</h2>
          <p>(closing soon...)</p>
        `;
        setTimeout(closePopup, 2500);
        break;

      case 'game':
        renderGame();
        break;
    }
  }

  // ---------- Game render: two-image swap for bowing ----------
  function renderGame() {
    content.innerHTML = `
      <div class="game-area">
        <div class="scene">
          <!-- Replace src with your actual image paths -->
          <img src="ash-legs.png" class="ash-legs" alt="Ash's legs">
          <img src="person-kneeling.png" class="person-img person-kneeling" id="person-kneeling" alt="Person kneeling">
          <img src="person-praise.png" class="person-img person-praise" id="person-praise" alt="Person raising arms in praise" style="display:none;">
        </div>
        <p class="instruction">CLICK (computer) or TAP (mobile) TO BOW BEFORE ASH!</p>
      </div>
    `;

    const kneelingImg = document.getElementById('person-kneeling');
    const praiseImg = document.getElementById('person-praise');
    const scene = document.querySelector('.scene');
    let bowTimeout = null;

    // Function to perform a bow
    function bow() {
      if (!kneelingImg || !praiseImg) return;
      // Show kneeling, hide praise
      kneelingImg.style.display = 'block';
      praiseImg.style.display = 'none';
      playCashSound();
      // Show floating money
      const money = document.createElement('div');
      money.className = 'floating-money';
      money.textContent = '-$1.00';
      scene.appendChild(money);
      money.addEventListener('animationend', () => money.remove());

      // After a short delay, show praise (sitting back up)
      clearTimeout(bowTimeout);
      bowTimeout = setTimeout(() => {
        kneelingImg.style.display = 'none';
        praiseImg.style.display = 'block';
      }, 500);
    }

    // Initially show praise image (person sitting up)
    kneelingImg.style.display = 'none';
    praiseImg.style.display = 'block';

    // Listen for clicks/taps anywhere in the game area
    const gameArea = document.querySelector('.game-area');
    if (gameArea) {
      gameArea.addEventListener('click', bow);
      gameArea.addEventListener('touchstart', (e) => {
        e.preventDefault();
        bow();
      });
    }
  }

  // Start the popup with initial screen
  render();

  // Close popup if user clicks backdrop (outside the inner box)
  popup.addEventListener('click', (e) => {
    if (e.target === popup) {
      closePopup();
    }
  });

})();
```
